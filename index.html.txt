        function update() {
            if (gameState !== 'PLAYING') return;

            let dx = 0;
            // 키보드 또는 터치 입력 확인
            if (keys.a || keys.ArrowLeft || touchControls.left) dx -= player.speed;
            if (keys.d || keys.ArrowRight || touchControls.right) dx += player.speed;

            if (dx !== 0) {
                player.x += dx;
                // 월드 경계 제한
                if (player.x < player.width/2) player.x = player.width/2;
                if (player.x > WORLD_WIDTH - player.width/2) player.x = WORLD_WIDTH - player.width/2;
                
                player.direction = dx > 0 ? 1 : -1;
                player.walkCycle += 0.2;
            } else {
                player.walkCycle = 0;
            }

            // 카메라 스무스 트래킹
            let targetCamX = player.x - canvas.width / 2;
            // 카메라 경계 제한
            if (targetCamX < 0) targetCamX = 0;
            if (targetCamX > WORLD_WIDTH - canvas.width) targetCamX = WORLD_WIDTH - canvas.width;
            
            camera.x += (targetCamX - camera.x) * 0.1;
        }

        // 심플한 캐릭터 그리기 (정면/측면 혼합 뷰)
        function drawPlayer(ctx, x, y, dir, walkCycle) {
            ctx.save();
            ctx.translate(x, y);

            // 스케일링으로 방향 전환
            ctx.scale(dir, 1);

            const legSwing = Math.sin(walkCycle) * 15;
            const armSwing = Math.cos(walkCycle) * 15;

            // 그림자
            ctx.fillStyle = 'rgba(0,0,0,0.15)';
            ctx.beginPath();
            ctx.ellipse(0, 0, 15, 5, 0, 0, Math.PI * 2);
            ctx.fill();

            // 뒷다리
            ctx.fillStyle = '#1f2937';
            ctx.save(); ctx.translate(3, -40); ctx.rotate((legSwing * Math.PI) / 180);
            ctx.fillRect(-6, 0, 12, 40); ctx.restore();

            // 몸통 (어두운 회색 수트 느낌)
            ctx.fillStyle = '#374151';
            ctx.fillRect(-15, -75, 30, 35);
            
            // 머리
            ctx.fillStyle = '#d4a373'; // 살구색
            ctx.beginPath(); ctx.arc(0, -90, 15, 0, Math.PI * 2); ctx.fill();
            
            // 앞다리
            ctx.fillStyle = '#111827';
            ctx.save(); ctx.translate(-3, -40); ctx.rotate((-legSwing * Math.PI) / 180);
            ctx.fillRect(-6, 0, 12, 40); ctx.restore();

            ctx.restore();
        }

        function draw() {
            // 배경 (벽)
            ctx.fillStyle = '#f8f9fa'; 
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            ctx.save();
            ctx.translate(-Math.floor(camera.x), 0); // y축 스크롤은 없음

            const responsiveFloorHeight = Math.min(150, canvas.height * 0.2);
            const floorY = canvas.height - responsiveFloorHeight;
            
            // 바닥 패턴 그리기 (세로 스트라이프 무늬)
            ctx.fillStyle = '#e6d5b8'; // 옅은 베이지
            ctx.fillRect(0, floorY, WORLD_WIDTH, responsiveFloorHeight);
            
            ctx.fillStyle = '#d5c4a1'; // 약간 진한 베이지
            const stripeWidth = 60;
            for (let i = 0; i < WORLD_WIDTH; i += stripeWidth * 2) {
                ctx.fillRect(i, floorY, stripeWidth, responsiveFloorHeight);
            }
            
            // 걸레받이 (벽과 바닥 사이 몰딩)
            ctx.fillStyle = '#5c4033'; // 짙은 갈색
            ctx.fillRect(0, floorY - 10, WORLD_WIDTH, 10);

            // 2. 전시 스팟 및 작품 그리기
            let closestArt = null;
            let closestSpot = null;
            let minDist = 150; // 상호작용 가능 거리 (캐릭터 중심과 액자 중심)

            const ART_Y_POS = floorY - Math.max(120, canvas.height * 0.3); // 모바일 해상도 고려 유동적 위치

            exhibitionSpots.forEach(spot => {
                const artX = spot.x;
                const artY = ART_Y_POS;

                const dist = Math.abs(player.x - artX);
                const isHovered = dist < minDist;

                const artwork = artworksData.find(a => a.spotId === spot.id);
                
                const maxW = Math.min(250, canvas.width * 0.6); 
                const maxH = Math.min(250, canvas.height * 0.4);
                let w = maxW; let h = maxH;

                if (artwork) {
                    if (isHovered) { closestArt = artwork; minDist = dist; }
                    
                    if (artwork.loadedImg) {
                        const aspect = artwork.imgW / artwork.imgH;
                        if (aspect > 1) { h = w / aspect; } else { w = h * aspect; }
                    }

                    // 액자 틀 (갈색 원목 느낌)
                    const frameBorder = 15;
                    ctx.shadowColor = 'rgba(0,0,0,0.3)';
                    ctx.shadowBlur = 10;
                    ctx.shadowOffsetY = 5;
                    ctx.fillStyle = '#4a3728'; // 짙은 갈색 액자
                    ctx.fillRect(artX - w/2 - frameBorder, artY - h/2 - frameBorder, w + frameBorder*2, h + frameBorder*2);
                    
                    // 매트보드 (그림 주변 흰 여백)
                    const matBorder = 25;
                    ctx.shadowBlur = 0;
                    ctx.shadowOffsetY = 0;
                    ctx.fillStyle = '#f0f0f0';
                    ctx.fillRect(artX - w/2 - matBorder, artY - h/2 - matBorder, w + matBorder*2, h + matBorder*2);

                    // 그림 그리기
                    if (artwork.loadedImg) {
                        ctx.drawImage(artwork.loadedImg, artX - w/2, artY - h/2, w, h);
                    } else {
                        ctx.fillStyle = '#ccc';
                        ctx.fillRect(artX - w/2, artY - h/2, w, h);
                    }
                    
                    // 작품 정보 라벨
                    const labelW = 120;
                    const labelH = 30;
                    ctx.fillStyle = 'rgba(255,255,255,0.9)';
                    ctx.shadowColor = 'rgba(0,0,0,0.1)';
                    ctx.shadowBlur = 4;
                    ctx.shadowOffsetY = 2;
                    ctx.fillRect(artX - labelW/2, artY + h/2 + matBorder + frameBorder + 10, labelW, labelH);
                    
                    ctx.shadowBlur = 0;
                    ctx.shadowOffsetY = 0;
                    ctx.font = 'bold 12px "Noto Serif KR"';
                    ctx.textAlign = 'center';
                    ctx.textBaseline = 'middle';
                    ctx.fillStyle = '#333';
                    
                    // 글자가 길면 자르기
                    let displayTitle = artwork.title;
                    if(ctx.measureText(displayTitle).width > labelW - 10) {
                        displayTitle = displayTitle.substring(0, 8) + '..';
                    }
                    ctx.fillText(displayTitle, artX, artY + h/2 + matBorder + frameBorder + 10 + labelH/2);

                    // 하이라이트 표시
                    if (isHovered) {
                        ctx.strokeStyle = '#4f46e5';
                        ctx.lineWidth = 3;
                        ctx.strokeRect(artX - w/2 - frameBorder - 5, artY - h/2 - frameBorder - 5, w + frameBorder*2 + 10, h + frameBorder*2 + 10);
                    }

                } else {
                    // 빈 공간
                    if (isHovered) { closestSpot = spot; minDist = dist; }

                    // 빈 스팟 가이드라인 표시
                    ctx.fillStyle = isHovered ? 'rgba(16, 185, 129, 0.1)' : 'transparent';
                    ctx.strokeStyle = isHovered ? '#10b981' : '#d1d5db';
                    ctx.lineWidth = 2;
                    ctx.setLineDash([5, 5]);
                    ctx.beginPath();
                    ctx.rect(artX - w/2, artY - h/2, w, h);
                    ctx.fill();
                    ctx.stroke();
                    ctx.setLineDash([]);
                    
                    if (isHovered) {
                        // 십자 마크
                        ctx.fillStyle = '#10b981';
                        ctx.fillRect(artX - 2, artY - 15, 4, 30);
                        ctx.fillRect(artX - 15, artY - 2, 30, 4);

                        ctx.textAlign = 'center';
                        ctx.fillText('작품 걸기', artX, artY + h/2 + 25);
                    }
                }
            });

            // 3. 플레이어 그리기
            drawPlayer(ctx, player.x, floorY, player.direction, player.walkCycle);

            ctx.restore();

            updateInteractionUI(closestArt, closestSpot);
        }

        function renderComments(commentsArray) {
            commentsListEl.innerHTML = '';
            if (commentsArray.length === 0) {
                commentsListEl.innerHTML = '<div class="text-center text-neutral-400 text-sm py-4">첫 번째 감상평을 남겨보세요.</div>';
                return;
            }
            
            // createdAt 기준으로 내림차순 정렬 (오프라인 모드 대응)
            commentsArray.sort((a,b) => b.createdAt - a.createdAt).forEach(data => {
                const date = new Date(data.createdAt).toLocaleDateString('ko-KR', { month: 'short', day: 'numeric', hour: '2-digit', minute:'2-digit' });
                
                const el = document.createElement('div');
                el.className = 'bg-white p-3 rounded-lg border border-neutral-100 shadow-sm flex flex-col gap-1';
                el.innerHTML = `
                    <div class="flex justify-between items-baseline">
                        <span class="font-bold text-sm text-neutral-800">${escapeHtml(data.author)}</span>
                        <span class="text-xs text-neutral-400">${date}</span>
                    </div>
                    <p class="text-sm text-neutral-600 break-words">${escapeHtml(data.text)}</p>
                `;
                commentsListEl.appendChild(el);
            });
        }

        function escapeHtml(unsafe) {
            return (unsafe||'').replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }

        uploadFile.addEventListener('change', (e) => {
            const file = e.target.files[0];
            if (!file) return;
            
            const reader = new FileReader();
            reader.onload = (event) => {
                const img = new Image();
                img.onload = () => {
                    let width = img.width;
                    let height = img.height;
                    const max_size = 1200; // 최대 해상도 제한
                    
                    if (width > height) {
                        if (width > max_size) { height *= max_size / width; width = max_size; }
                    } else {
                        if (height > max_size) { width *= max_size / height; height = max_size; }
                    }

                    const compCanvas = document.getElementById('compress-canvas');
                    compCanvas.width = width; compCanvas.height = height;
                    const ctx = compCanvas.getContext('2d');
                    ctx.drawImage(img, 0, 0, width, height);

                    let quality = 0.8;
                    let base64Data = compCanvas.toDataURL('image/jpeg', quality);
                    
                    // 크기가 너무 크면 품질을 낮춰가며 재시도
                    while (base64Data.length > 1300000 && quality > 0.1) {
                        quality -= 0.1;
                        base64Data = compCanvas.toDataURL('image/jpeg', quality);
                    }

                    currentCompressedImageBase64 = base64Data;
                    uploadPreview.src = base64Data;
                    uploadPreviewContainer.classList.remove('hidden');
                };
                img.src = event.target.result;
            };
            reader.readAsDataURL(file);
        });

            genLoadingSpinner.classList.remove('hidden');
            uploadPreviewContainer.classList.remove('hidden');
            btnGenerateImg.disabled = true;

            try {
                // imagen-4.0-generate-001 모델 사용 (가이드라인 권장)
                const payload = { 
                    instances: { prompt: prompt }, 
                    parameters: { sampleCount: 1 } 
                };
                const apiKey = ""; 
                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${apiKey}`;
                
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                
                if (!response.ok) throw new Error("API Request Failed");

                const result = await response.json();
                
                if (result.predictions && result.predictions.length > 0 && result.predictions[0].bytesBase64Encoded) {
                    currentCompressedImageBase64 = `data:image/jpeg;base64,${result.predictions[0].bytesBase64Encoded}`;
                    uploadPreview.src = currentCompressedImageBase64;
                } else {
                    throw new Error("Invalid response format");
                }
            } catch (error) {
                console.error("이미지 생성 에러:", error);
                showToast("이미지 생성에 실패했습니다. 프롬프트를 바꿔 다시 시도해주세요.");
            } finally {
                genLoadingSpinner.classList.add('hidden');
                btnGenerateImg.disabled = false;
            }
        });


        // --- Gemini API: AI 도슨트 (이미지 분석) ---
        aiDocentBtn.addEventListener('click', async () => {
            if (!activeArtwork || !activeArtwork.imageBase64) return;
            
            aiDocentBtn.disabled = true;
            aiDocentBtn.innerHTML = `<svg class="animate-spin h-4 w-4 mr-2 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24"><circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle><path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path></svg> 작품 분석 중...`;
            aiDocentResult.classList.add('hidden');
            
            try {
                // base64 데이터 정제
                const base64Data = activeArtwork.imageBase64.split(',')[1];
                const mimeType = activeArtwork.imageBase64.match(/[^:]\w+\/[\w-+\d.]+(?=;|,)/)[0] || 'image/jpeg';
                
                const prompt = `당신은 친절하고 전문적인 미술관 도슨트입니다. 다음 작품의 제목은 "${activeArtwork.title}" 입니다. 
                이 이미지를 보고 관람객에게 들려줄 만한 흥미롭고 깊이 있는 해설을 3~4문장 정도로 자연스럽게 작성해주세요. 화풍, 색감, 연상되는 분위기 등을 포함하면 좋습니다.`;
                
                const payload = {
                    contents: [
                        {
                            role: "user",
                            parts: [
                                { text: prompt },
                                { inlineData: { mimeType: mimeType, data: base64Data } }
                            ]
                        }
                    ],
                };
                
                const apiKey = ""; 
                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;
                
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });
                
                const result = await response.json();
                
                if (result.candidates && result.candidates.length > 0) {
                    const text = result.candidates[0].content.parts[0].text;
                    aiDocentResult.textContent = text;
                    aiDocentResult.classList.remove('hidden');
                } else {
                    throw new Error("No response from AI");
                }
            } catch(e) {
                console.error("AI 도슨트 에러:", e);
                showToast("도슨트를 불러오는 데 실패했습니다.");
            } finally {
                aiDocentBtn.disabled = false;
                aiDocentBtn.innerHTML = `<svg xmlns="http://www.w3.org/2000/svg" width="18" height="18" fill="currentColor" viewBox="0 0 16 16"><path d="M8 16A8 8 0 1 0 8 0a8 8 0 0 0 0 16zm.93-9.412-1 4.705c-.07.34.029.533.304.533.194 0 .487-.07.686-.246l-.088.416c-.287.346-.92.598-1.465.598-.703 0-1.002-.422-.808-1.319l.738-3.468c.064-.293.006-.399-.287-.47l-.451-.081.082-.381 2.29-.287zM8 5.5a1 1 0 1 1 0-2 1 1 0 0 1 0 2z"/></svg><span>AI 도슨트에게 작품 해설 듣기</span>`;
            }
        });


        uploadForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            if (!activeWallSpot || !currentCompressedImageBase64) {
                showToast("이미지를 선택하거나 생성해주세요."); return;
            }

            const title = document.getElementById('upload-title').value.trim();
            const artist = document.getElementById('upload-artist').value.trim();

            uploadSpinner.classList.remove('hidden');
            uploadBtnText.textContent = '업로드 중...';
            submitUploadBtn.disabled = true;

            const success = await saveArtwork(title, artist, currentCompressedImageBase64, activeWallSpot.id);
            
            uploadSpinner.classList.add('hidden');
            uploadBtnText.textContent = '전시하기';
            submitUploadBtn.disabled = false;

            if (success) {
                showToast("작품이 성공적으로 전시되었습니다!");
                closeUploadModal();
            }
        });

        function handleAction() {
            if (activeArtwork) openViewModal(activeArtwork);
            else if (activeWallSpot) openUploadModal(activeWallSpot);
        }

        function openViewModal(artwork) {
            gameState = 'VIEWING';
            artTitleEl.textContent = artwork.title; 
            artArtistEl.textContent = artwork.artist;
            loadingSpinner.style.display = 'flex'; 
            artImageEl.style.opacity = '0';
            aiDocentResult.classList.add('hidden'); // 모달 열때 도슨트 결과 숨김
            
            if (artwork.loadedImg) {
                artImageEl.src = artwork.loadedImg.src; 
                loadingSpinner.style.display = 'none'; 
                artImageEl.style.opacity = '1';
            }
            loadComments(artwork.id);
            
            backdropEl.classList.remove('hidden'); 
            modalEl.classList.remove('hidden'); 
            modalEl.classList.add('modal-active');
            promptEl.classList.add('hidden'); 
            uploadPromptEl.classList.add('hidden');
            for(let key in keys) keys[key] = false;
        }

        function closeViewModal() {
            gameState = 'PLAYING';
            backdropEl.classList.add('hidden'); 
            modalEl.classList.add('hidden'); 
            modalEl.classList.remove('modal-active');
            if (unsubscribeComments) unsubscribeComments(); 
            setTimeout(()=> { artImageEl.src = ''; }, 300); // 닫힌 후 이미지 리셋
        }

        function openUploadModal(spot) {
            gameState = 'UPLOADING';
            activeWallSpot = spot; 
            uploadForm.reset(); 
            currentCompressedImageBase64 = null;
            uploadPreviewContainer.classList.add('hidden'); 
            uploadPreview.src = '';
            aiPromptInput.value = '';
            
            uploadBackdropEl.classList.remove('hidden'); 
            uploadModalEl.classList.remove('hidden'); 
            uploadModalEl.classList.add('modal-active');
            promptEl.classList.add('hidden'); 
            uploadPromptEl.classList.add('hidden');
            for(let key in keys) keys[key] = false;
        }

        function closeUploadModal() {
            gameState = 'PLAYING';
            uploadBackdropEl.classList.add('hidden'); 
            uploadModalEl.classList.add('hidden'); 
            uploadModalEl.classList.remove('modal-active');
            activeWallSpot = null;
        }

        // 모달 닫기 이벤트 바인딩
        closeBtn.addEventListener('click', closeViewModal);
        backdropEl.addEventListener('click', (e) => { if(e.target === backdropEl) closeViewModal(); });
        cancelUploadBtn.addEventListener('click', closeUploadModal);
        uploadBackdropEl.addEventListener('click', (e) => { if(e.target === uploadBackdropEl) closeUploadModal(); });

        // 메인 게임 루프
        function gameLoop() {
            update();
            draw();
            requestAnimationFrame(gameLoop);
        }
        gameLoop();

    </script>
